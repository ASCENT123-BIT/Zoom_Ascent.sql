FROM trips 
WHERE status = 'completed'

# Q4a- Find the duplicate trips?
SELECT customer_id, driver_id, trip_date, fare,
MIN(trip_id) AS trip_id_1,
MAX(trip_id) AS trip_id_2,
COUNT(*) AS duplicate_count
FROM trips
GROUP BY customer_id, driver_id, trip_date, fare
HAVING COUNT(*) > 1;


# Q4b-  How many Completed trips have a missing fare?
SELECT COUNT(*) AS missing_fare 
FROM trips 
WHERE status = 'completed' AND fare is NULL;

#Q6- Revenue by City?
SELECT city,
COUNT(*) AS number_of_trips,
SUM(fare) AS revenue, 
ROUND(AVG(fare), 2) AS average_fare
FROM trips
WHERE status= 'Completed'
GROUP BY city
ORDER BY revenue DESC;


#Q7-  Revenue by Month?
SELECT
DATE_FORMAT(trip_date, '%Y-%m') AS month,
COUNT(*) AS number_of_trips,
SUM(fare) AS revenue
FROM trips
WHERE status= 'Completed'
GROUP BY Month
ORDER BY revenue DESC;


#Q8-  Revenue by Vehicle Type?
SELECT drivers.vehicle_type,
COUNT(*) AS number_of_trips,
SUM(fare) AS revenue
FROM trips
INNER JOIN drivers ON trips.driver_id = drivers.driver_id
WHERE trips.status= 'Completed'
GROUP BY drivers.vehicle_type
ORDER BY revenue DESC;


#Q9- Which customers have never booked a trip?
SELECT customers.customer_name,
COUNT(trips.trip_id) AS number_of_trips
FROM customers
LEFT JOIN trips ON customers.customer_id = trips.customer_id
WHERE trips.trip_id IS NULL
GROUP BY customers.customer_name;


#Q10- Who are the top 3 customers by total spend on Completed trips?
SELECT customers.customer_name,
COUNT(trips.trip_id) AS number_of_trips,
SUM(fare) AS total_spend
FROM customers
LEFT JOIN trips ON customers.customer_id = trips.customer_id
WHERE trips.status = 'Completed'
GROUP BY customers.customer_name
ORDER BY total_spend DESC
LIMIT 3;
